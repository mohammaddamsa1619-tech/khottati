from flask import Flask, request, jsonify
import openai

app = Flask(__name__)
openai.api_key = "حط مفتاح OpenAI هون - sk-..."

@app.route('/api/generate', methods=['POST'])
def generate():
    lesson = request.json['lesson']
    
    prompt = f"""
    انت معلم أردني للصف الأول. ولد حزمة كاملة لدرس: {lesson}
    المطلوب 6 أشياء بصيغة JSON:
    1. worksheet: 5 تمارين جمع برسومات
    2. explanation: كيف يشرح المعلم الدرس بـ 3 جمل
    3. summary: تلخيص للطالب
    4. infographic_prompt: وصف لصورة انفوجرافيك ل Midjourney
    5. mindmap_text: نص الخريطة الذهنية
    6. flashcards: 8 بطاقات جمع
    """
    
    # هنا بنادي GPT-4
    response = openai.ChatCompletion.create(
        model="gpt-4",
        messages=[{"role":"user", "content": prompt}]
    )
    
    # وبعدين بنحول النص لـ PDF باستخدام reportlab
    # ونرجع روابط التحميل
    
    return jsonify({
        "worksheet": "/files/worksheet.pdf",
        "summary": "/files/summary.pdf", 
        "infographic": "/files/infographic.png",
        "mindmap": "/files/mindmap.png",
        "flashcards": "/files/flashcards.pdf",
        "explanation": "شرح مبسط..."
    })

if __name__ == '__main__':
    app.run()
