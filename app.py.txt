from flask import Flask, request, jsonify, send_from_directory
import os
from openai import OpenAI

app = Flask(__name__, static_folder='.')

# مفتاح OpenAI من متغير البيئة في Render
client = None
api_key = os.environ.get("OPENAI_API_KEY")
if api_key:
    client = OpenAI(api_key=api_key)

@app.route('/')
def home():
    return send_from_directory('.', 'index.html')

@app.route('/api/generate', methods=['POST'])
def generate():
    data = request.get_json()
    lesson = data.get('lesson', 'درس الجمع')

    explanation = f"شرح مبسط لدرس {lesson} بثلاث خطوات عملية للأطفال."
    summary = f"تلخيص درس {lesson}"

    # اذا في مفتاح OpenAI، بنادي GPT
    if client:
        try:
            prompt = f"انت معلم أردني للصف الأول. اكتب شرح مبسط جدا لدرس: {lesson} في 3 جمل قصيرة بالعربي."
            response = client.chat.completions.create(
                model="gpt-4o-mini",
                messages=[{"role":"user", "content": prompt}]
            )
            explanation = response.choices[0].message.content
        except Exception as e:
            explanation = f"شرح لدرس {lesson}: {str(e)}"

    return jsonify({
        "worksheet": "#",
        "summary": "#",
        "infographic": "#",
        "mindmap": "#",
        "flashcards": "#",
        "explanation": explanation,
        "summary_text": summary
    })

if __name__ == '__main__':
    app.run(host="0.0.0.0", port=10000)
